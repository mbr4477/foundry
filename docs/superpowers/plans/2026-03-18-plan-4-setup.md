# foundry-setup Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `foundry-setup`, an idempotent one-shot CLI that initializes a Gitea instance for Foundry: creates the bot user, generates an API token, and registers a system-level webhook.

**Architecture:** A `clap`-derived CLI with a single `run` command. Each setup step is a separate function that checks before creating (idempotent). A `GiteaAdminClient` wraps `reqwest` for all calls requiring admin privileges. The bot token (requiring the `--admin-token` to generate) is printed once and never stored.

**Tech Stack:** Rust, `clap`, `reqwest`, `serde_json`, `mockito` (tests)

```rust
// foundry-setup/src/client.rs
use reqwest::{Client, StatusCode};
use serde::{Deserialize, Serialize};
use thiserror::Error;

#[derive(Debug, Error)]
pub enum SetupError {
    #[error("HTTP error: {0}")]
    Http(#[from] reqwest::Error),
    #[error("Unauthorized — check admin token")]
    Unauthorized,
    #[error("Unexpected response {status}: {body}")]
    Unexpected { status: u16, body: String },
}

#[derive(Debug, Deserialize)]
pub struct GiteaUser {
    pub id: u64,
    pub login: String,
}

#[derive(Debug, Deserialize)]
pub struct GiteaToken {
    pub id: u64,
    pub name: String,
    pub sha1: String,
}

#[derive(Debug, Deserialize)]
pub struct GiteaHook {
    pub id: u64,
    pub config: HookConfig,
}

#[derive(Debug, Deserialize)]
pub struct HookConfig {
    pub url: String,
}

pub struct GiteaAdminClient {
    base_url: String,
    admin_token: String,
    http: Client,
}

impl GiteaAdminClient {
    pub fn new(base_url: String, admin_token: String) -> Self {
        Self {
            base_url: base_url.trim_end_matches('/').to_string(),
            admin_token,
            http: Client::new(),
        }
    }

    fn api(&self, path: &str) -> String {
        format!("{}/api/v1{}", self.base_url, path)
    }

    async fn check(&self, resp: reqwest::Response) -> Result<reqwest::Response, SetupError> {
        if resp.status() == StatusCode::UNAUTHORIZED {
            return Err(SetupError::Unauthorized);
        }
        if !resp.status().is_success() {
            let status = resp.status().as_u16();
            let body = resp.text().await.unwrap_or_default();
            return Err(SetupError::Unexpected { status, body });
        }
        Ok(resp)
    }

    pub async fn get_version(&self) -> Result<String, SetupError> {
        let resp = self.http
            .get(self.api("/version"))
            .bearer_auth(&self.admin_token)
            .send().await?;
        let json: serde_json::Value = self.check(resp).await?.json().await?;
        Ok(json["version"].as_str().unwrap_or("unknown").to_string())
    }

    pub async fn user_exists(&self, username: &str) -> Result<bool, SetupError> {
        let resp = self.http
            .get(self.api(&format!("/users/{}", username)))
            .bearer_auth(&self.admin_token)
            .send().await?;
        match resp.status() {
            StatusCode::OK => Ok(true),
            StatusCode::NOT_FOUND => Ok(false),
            StatusCode::UNAUTHORIZED => Err(SetupError::Unauthorized),
            s => {
                let body = resp.text().await.unwrap_or_default();
                Err(SetupError::Unexpected { status: s.as_u16(), body })
            }
        }
    }

    pub async fn create_user(&self, username: &str, email: &str, password: &str)
        -> Result<GiteaUser, SetupError>
    {
        #[derive(Serialize)]
        struct Body<'a> {
            username: &'a str,
            email: &'a str,
            password: &'a str,
            must_change_password: bool,
            send_notify: bool,
        }
        let resp = self.http
            .post(self.api("/admin/users"))
            .bearer_auth(&self.admin_token)
            .json(&Body { username, email, password, must_change_password: false, send_notify: false })
            .send().await?;
        Ok(self.check(resp).await?.json().await?)
    }

    pub async fn list_user_tokens(&self, username: &str)
        -> Result<Vec<GiteaToken>, SetupError>
    {
        let resp = self.http
            .get(self.api(&format!("/users/{}/tokens", username)))
            .bearer_auth(&self.admin_token)
            .send().await?;
        Ok(self.check(resp).await?.json().await?)
    }

    pub async fn create_user_token(&self, username: &str, token_name: &str)
        -> Result<GiteaToken, SetupError>
    {
        #[derive(Serialize)]
        struct Body<'a> {
            name: &'a str,
        }
        let resp = self.http
            .post(self.api(&format!("/users/{}/tokens", username)))
            .bearer_auth(&self.admin_token)
            .json(&Body { name: token_name })
            .send().await?;
        Ok(self.check(resp).await?.json().await?)
    }

    pub async fn find_system_webhook(&self, webhook_url: &str)
        -> Result<Option<u64>, SetupError>
    {
        let resp = self.http
            .get(self.api("/admin/hooks"))
            .bearer_auth(&self.admin_token)
            .send().await?;
        let hooks: Vec<GiteaHook> = self.check(resp).await?.json().await?;
        Ok(hooks.into_iter()
            .find(|h| h.config.url == webhook_url)
            .map(|h| h.id))
    }

    pub async fn create_system_webhook(
        &self,
        webhook_url: &str,
        secret: &str,
    ) -> Result<u64, SetupError> {
        #[derive(Serialize)]
        struct Body<'a> {
            #[serde(rename = "type")]
            hook_type: &'a str,
            config: HookBodyConfig<'a>,
            events: Vec<&'a str>,
            active: bool,
        }
        #[derive(Serialize)]
        struct HookBodyConfig<'a> {
            url: &'a str,
            content_type: &'a str,
            secret: &'a str,
        }
        let resp = self.http
            .post(self.api("/admin/hooks"))
            .bearer_auth(&self.admin_token)
            .json(&Body {
                hook_type: "gitea",
                config: HookBodyConfig {
                    url: webhook_url,
                    content_type: "json",
                    secret,
                },
                events: vec!["issues", "issue_comment", "pull_request", "pull_request_review"],
                active: true,
            })
            .send().await?;
        let hook: serde_json::Value = self.check(resp).await?.json().await?;
        Ok(hook["id"].as_u64().unwrap_or(0))
    }

    pub async fn update_system_webhook(
        &self,
        hook_id: u64,
        webhook_url: &str,
        secret: &str,
    ) -> Result<(), SetupError> {
        #[derive(Serialize)]
        struct Body<'a> {
            config: HookBodyConfig<'a>,
            events: Vec<&'a str>,
            active: bool,
        }
        #[derive(Serialize)]
        struct HookBodyConfig<'a> {
            url: &'a str,
            content_type: &'a str,
            secret: &'a str,
        }
        let resp = self.http
            .patch(self.api(&format!("/admin/hooks/{}", hook_id)))
            .bearer_auth(&self.admin_token)
            .json(&Body {
                config: HookBodyConfig {
                    url: webhook_url,
                    content_type: "json",
                    secret,
                },
                events: vec!["issues", "issue_comment", "pull_request", "pull_request_review"],
                active: true,
            })
            .send().await?;
        self.check(resp).await?;
        Ok(())
    }
}
```

```rust
// foundry-setup/src/steps.rs
use crate::client::{GiteaAdminClient, SetupError};

#[derive(Debug, PartialEq)]
pub enum EnsureResult {
    Created,
    AlreadyExists,
}

#[derive(Debug)]
pub enum WebhookResult {
    Created(u64),
    Updated(u64),
    AlreadyExists(u64),
}

#[derive(Debug)]
pub enum TokenResult {
    Created(String),
    AlreadyExists,
}

pub async fn ensure_user(
    client: &GiteaAdminClient,
    username: &str,
    email: &str,
) -> Result<EnsureResult, SetupError> {
    if client.user_exists(username).await? {
        return Ok(EnsureResult::AlreadyExists);
    }
    // Generate a random password — user will never log in interactively
    let password = format!("Foundry-{}", uuid_v4_simple());
    client.create_user(username, email, &password).await?;
    Ok(EnsureResult::Created)
}

pub async fn ensure_token(
    client: &GiteaAdminClient,
    username: &str,
    token_name: &str,
) -> Result<TokenResult, SetupError> {
    let tokens = client.list_user_tokens(username).await?;
    if tokens.iter().any(|t| t.name == token_name) {
        return Ok(TokenResult::AlreadyExists);
    }
    let token = client.create_user_token(username, token_name).await?;
    Ok(TokenResult::Created(token.sha1))
}

pub async fn ensure_webhook(
    client: &GiteaAdminClient,
    webhook_url: &str,
    secret: &str,
) -> Result<WebhookResult, SetupError> {
    if let Some(existing_id) = client.find_system_webhook(webhook_url).await? {
        client.update_system_webhook(existing_id, webhook_url, secret).await?;
        return Ok(WebhookResult::Updated(existing_id));
    }
    let id = client.create_system_webhook(webhook_url, secret).await?;
    Ok(WebhookResult::Created(id))
}

/// Generate a simple random-looking string for the initial bot password.
/// Not cryptographically strong — just needs to be unique enough to set.
fn uuid_v4_simple() -> String {
    use std::time::{SystemTime, UNIX_EPOCH};
    let nanos = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|d| d.subsec_nanos())
        .unwrap_or(12345);
    format!("{:08x}-setup", nanos)
}
```

```rust
// foundry-setup/src/main.rs
mod client;
mod steps;

use clap::Parser;
use steps::{EnsureResult, TokenResult, WebhookResult};

#[derive(Parser)]
#[command(
    name = "foundry-setup",
    about = "Initialize Gitea for Foundry (idempotent)"
)]
struct Args {
    /// Gitea base URL
    #[arg(long, env = "GITEA_URL")]
    gitea_url: String,

    /// Gitea admin token (requires admin scope)
    #[arg(long, env = "FOUNDRY_ADMIN_TOKEN")]
    admin_token: String,

    /// Webhook URL that Gitea will POST events to
    #[arg(long)]
    webhook_url: String,

    /// Webhook HMAC secret (also set as FOUNDRY_WEBHOOK_SECRET in foundryd config)
    #[arg(long, env = "FOUNDRY_WEBHOOK_SECRET")]
    webhook_secret: String,

    /// Username for the bot account
    #[arg(long, default_value = "foundry-bot")]
    bot_username: String,

    /// Email for the bot account
    #[arg(long, default_value = "foundry-bot@gitea.local")]
    bot_email: String,
}

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    let args = Args::parse();
    let client = client::GiteaAdminClient::new(args.gitea_url.clone(), args.admin_token.clone());

    println!("=== foundry-setup ===");
    println!("Gitea: {}", args.gitea_url);
    println!();

    // Step 1: Verify connectivity
    print!("[1/3] Verifying Gitea connectivity... ");
    let version = client.get_version().await
        .map_err(|e| anyhow::anyhow!("Cannot reach Gitea at {}: {}", args.gitea_url, e))?;
    println!("OK (Gitea {})", version);

    // Step 2: Ensure bot user exists
    print!("[2/3] Bot user @{}... ", args.bot_username);
    match steps::ensure_user(&client, &args.bot_username, &args.bot_email).await? {
        EnsureResult::Created => println!("created"),
        EnsureResult::AlreadyExists => println!("already exists (skipped)"),
    }

    // Step 3: Ensure bot API token
    print!("      API token 'foundry'... ");
    match steps::ensure_token(&client, &args.bot_username, "foundry").await? {
        TokenResult::Created(token) => {
            let redacted = if token.len() > 4 {
                format!("...{}", &token[token.len()-4..])
            } else {
                "****".into()
            };
            println!("created ({})", redacted);
            println!();
            println!("  *** SAVE THIS TOKEN — it will not be shown again ***");
            println!("  FOUNDRY_GITEA_TOKEN={}", token);
            println!();
        }
        TokenResult::AlreadyExists => println!("already exists (skipped)"),
    }

    // Step 4: Ensure system webhook
    print!("[3/3] System webhook {}... ", args.webhook_url);
    match steps::ensure_webhook(&client, &args.webhook_url, &args.webhook_secret).await? {
        WebhookResult::Created(id) => println!("created (id={})", id),
        WebhookResult::Updated(id) => println!("updated (id={})", id),
        WebhookResult::AlreadyExists(id) => println!("already exists (id={}, skipped)", id),
    }

    println!();
    println!("=== Setup complete ===");
    println!("Add @{} as a collaborator on any repos you want Foundry to manage.", args.bot_username);
    Ok(())
}
```
